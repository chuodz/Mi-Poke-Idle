// Tus Pokémon disponibles (almacén / Pokédex) y tu equipo activo en batalla let playerTeam = [ { id: 1, name: "Pikachu", level: 50, active: true }, { id: 4, name: "Charmander", level: 6, active: true } ]; let pcStorage = [ { id: 25, name: "Bulbasaur", level: 10, active: false }, { id: 37, name: "Vulpix", level: 12, active: false } ]; // Función para cambiar un Pokémon del equipo por uno del almacenamiento function swapPokemon(indexInTeam, storageId) { let teamMember = playerTeam[indexInTeam]; let storageIndex = pcStorage.findIndex(p => p.id === storageId); if (storageIndex !== -1) { let newMember = pcStorage[storageIndex]; // Intercambiar estados teamMember.active = false; newMember.active = true; // Actualizar los arrays pcStorage[storageIndex] = teamMember; playerTeam[indexInTeam] = newMember; console.log(`¡Has cambiado a ${teamMember.name} por ${newMember.name} en tu equipo!`); renderTeamUI(); // Actualiza los cuadros superiores de la pantalla } }
